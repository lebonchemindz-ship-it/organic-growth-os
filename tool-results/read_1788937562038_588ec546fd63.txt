    72→    72→    data: {
    73→    73→      keywordsTracked: kw,
    74→    74→      keywordsTop10: kwTop10,

    88→    88→async function listKeywords(args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {
    89→    89→  const limit = Math.min(Number(args.limit) || 12, 30)
    90→    90→  // Funnel through the keywords route (it owns the materialized GSC universe;
    91→    91→  // this function's local DB may be empty on a cold instance).
    92→    92→  if (ctx.appOrigin) {
    93→    93→    try {
    94→    94→      const res = await fetch(`${ctx.appOrigin}/api/keywords?brand=${encodeURIComponent(ctx.brandSlug)}`, {
    95→    95→        signal: AbortSignal.timeout(50_000),
    96→    96→      })
    97→    97→      if (res.ok) {
    98→    98→        const json = (await res.json()) as { keywords?: Array<Record<string, unknown>>; summary?: Record<string, number> }
    99→    99→        if (Array.isArray(json.keywords) && json.keywords.length > 0) {
   100→   100→          let kws = json.keywords
   101→   101→          if (typeof args.intent === 'string' && args.intent) kws = kws.filter((k) => k.intent === String(args.intent).toUpperCase())
   102→   102→          if (typeof args.funnel === 'string' && args.funnel) kws = kws.filter((k) => k.funnelStage === String(args.funnel).toUpperCase())
   103→   103→          if (args.maxPosition) kws = kws.filter((k) => Number(k.currentPosition) > 0 && Number(k.currentPosition) <= Number(args.maxPosition))
   104→   104→          const s = json.summary || {}
   105→   105→          return {
   106→   106→            ok: true,
   107→   107→            summary: `${kws.length} keywords (real GSC + research data, tracked positions first) — universe: ${s.total ?? kws.length} total, ${s.live ?? 0} live, ${s.top10 ?? 0} in top 10`,
   108→   108→            data: kws.slice(0, limit).map((k) => ({
   109→   109→              term: k.term, intent: k.intent, funnel: k.funnelStage, volume: k.monthlyVolume,
   110→   110→              difficulty: k.difficulty, position: k.currentPosition || null,
   111→   111→              previous: k.previousPosition || null,
   112→   112→              delta: k.currentPosition && k.previousPosition ? Number(k.previousPosition) - Number(k.currentPosition) : null,
   113→   113→              source: k.source,
   114→   114→            })),
   115→   115→          }
   116→   116→        }
   117→   117→      }
   118→   118→    } catch (e) {
   119→   119→      console.error('[assistant:list_keywords] funnel failed:', e instanceof Error ? e.message : e)
   120→   120→      // fall through to the local DB
   121→   121→    }
   122→   122→  }
   123→   123→  const where: Record<string, unknown> = { brandId: ctx.brandId }
   124→   124→  if (typeof args.intent === 'string' && args.intent) where.intent = args.intent.toUpperCase()
   125→   125→  if (typeof args.funnel === 'string' && args.funnel) where.funnelStage = args.funnel.toUpperCase()
   126→   126→  if (args.maxPosition) where.currentPosition = { gt: 0, lte: Number(args.maxPosition) }
   127→   127→  const kws = await db.keyword.findMany({
   128→   128→    where,
   129→   129→    orderBy: [{ currentPosition: 'asc' }, { monthlyVolume: 'desc' }],
   130→   130→    take: limit,
   131→   131→  })
   132→   132→  return {
   133→   133→    ok: true,
   134→   134→    summary: `${kws.length} keywords (tracked positions first)`,
   135→   135→    data: kws.map(k => ({
   136→   136→      term: k.term, intent: k.intent, funnel: k.funnelStage, volume: k.monthlyVolume,
   137→   137→      difficulty: k.difficulty, position: k.currentPosition || null,
   138→   138→      previous: k.previousPosition || null,
   139→   139→      delta: k.currentPosition && k.previousPosition ? k.previousPosition - k.currentPosition : null,
   140→   140→    })),
   141→   141→  }
   142→   142→}

   256→   256→    data: {

   258→   258→      title,
   259→   259→      description: String(args.description || ''),
   260→   260→      type,
   261→   261→      priority,
   262→   262→      autonomy: priority === 'HIGH' ? 'YELLOW' : 'GREEN',
   263→   263→      status: 'QUEUED',
   264→   264→      source: 'ASSISTANT',
   265→   265→    },
   266→   266→  })
   267→   267→  await logEvent(ctx, 'TASK_CREATED', `Agent queued task: ${title}`, `type=${type} priority=${priority}`)
   268→   268→  return { ok: true, summary: `Task queued (id ${task.id.slice(-6)}): ${title}`, data: { id: task.id, title, type, priority, status: 'QUEUED' } }

   271→   271→async function updateTask(args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {

   273→   273→  const status = String(args.status || '').toUpperCase()

   275→   275→    return { ok: false, summary: 'update_task requires id and a valid status (QUEUED|RUNNING|DONE|FAILED).' }

   280→   280→    where: { id },
   281→   281→    data: {
   282→   282→      status,
   283→   283→      result: args.result !== undefined ? String(args.result) : existing.result,
   284→   284→      completedAt: status === 'DONE' || status === 'FAILED' ? new Date() : null,
   285→   285→    },
   286→   286→  })
   287→   287→  await logEvent(ctx, 'TASK_UPDATED', `Agent set task "${task.title}" → ${status}`, String(args.result || ''))

   291→   291→async function addKeywords(args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {
   292→   292→  const terms = Array.isArray(args.terms) ? args.terms.slice(0, 20) : []
   293→   293→  if (!terms.length) return { ok: false, summary: 'add_keywords requires terms: array.' }
   294→   294→  const created: string[] = []
   295→   295→  for (const t of terms) {
   296→   296→    const term = typeof t === 'string' ? t : String((t as Record<string, unknown>).term || '').trim()
   297→   297→    if (!term) continue
   298→   298→    const exists = await db.keyword.findFirst({ where: { brandId: ctx.brandId, term } })
   299→   299→    if (exists) continue
   300→   300→    const kw = await db.keyword.create({
   301→   301→      data: {
   302→   302→        brandId: ctx.brandId,
   303→   303→        term,
   304→   304→        intent: typeof t === 'object' && (t as Record<string, unknown>).intent ? String((t as Record<string, unknown>).intent).toUpperCase() : 'INFORMATIONAL',
   305→   305→        funnelStage: typeof t === 'object' && (t as Record<string, unknown>).funnel ? String((t as Record<string, unknown>).funnel).toUpperCase() : 'TOFU',
   306→   306→        monthlyVolume: Number(typeof t === 'object' ? (t as Record<string, unknown>).volume : 0) || 0,
   307→   307→        difficulty: Number(typeof t === 'object' ? (t as Record<string, unknown>).difficulty : 0) || 0,
   308→   308→        currentPosition: 0,
   309→   309→        previousPosition: 0,
   310→   310→        commercialValue: 30,
   311→   311→        aeoValue: 40,
   312→   312→        geoValue: 30,
   313→   313→        status: 'TRACKING',
   314→   314→        source: 'AGENT',
   315→   315→      },
   316→   316→    })
   317→   317→    created.push(kw.term)
   318→   318→  }
   319→   319→  await logEvent(ctx, 'KEYWORDS_ADDED', `Agent added ${created.length} keywords to the universe`, created.join(', '))
   320→   320→  return { ok: true, summary: `${created.length} keywords added (duplicates skipped).`, data: { added: created } }
   321→   321→}

   323→   323→// Real keyword research via DataForSEO (when the credentials work).
   324→   324→// Returns actionable guidance when they don't. Writes are funneled through
   325→   325→// the keywords route so the dashboard sees them (separate serverless
   326→   326→// functions have separate ephemeral DBs).
   327→   327→async function researchKeywordsTool(args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {
   328→   328→  const seed = String(args.seed || '').trim()
   329→   329→  if (!seed) return { ok: false, summary: 'research_keywords requires a seed keyword.' }
   330→   330→  const limit = Math.min(Number(args.limit) || 20, 25)
   331→   331→
   332→   332→  if (ctx.appOrigin) {
   333→   333→    try {
   334→   334→      const res = await fetch(`${ctx.appOrigin}/api/keywords`, {
   335→   335→        method: 'POST',
   336→   336→        headers: { 'content-type': 'application/json' },
   337→   337→        body: JSON.stringify({ action: 'research', seed, limit, brandSlug: ctx.brandSlug }),
   338→   338→        signal: AbortSignal.timeout(55_000),
   339→   339→      })
   340→   340→      const json = (await res.json()) as { ok?: boolean; message?: string; error?: string; addedCount?: number; suggestions?: Array<{ term: string; volume: number; difficulty: number }> }
   341→   341→      if (res.ok && json.ok) {
   342→   342→        await logEvent(ctx, 'KEYWORDS_RESEARCHED', `DataForSEO research on "${seed}": ${json.addedCount ?? 0} real keywords added`, (json.suggestions || []).slice(0, 10).map(s => `${s.term} (${s.volume}/mo, KD ${s.difficulty})`).join(', '))
   343→   343→        return {
   344→   344→          ok: true,
   345→   345→          summary: `Researched "${seed}" via DataForSEO — ${(json.suggestions || []).length} suggestions, ${json.addedCount ?? 0} added to the universe with real volumes.`,
   346→   346→          data: { seed, added: json.addedCount, suggestions: (json.suggestions || []).slice(0, 10) },
   347→   347→        }
   348→   348→      }
   349→   349→      return { ok: false, summary: `Keyword research unavailable: ${json.message || json.error || `HTTP ${res.status}`}` }
   350→   350→    } catch (e) {
   351→   351→      // fall through to the direct lib path
   352→   352→      console.error('[assistant:research] funnel failed:', e instanceof Error ? e.message : e)
   353→   353→    }
   354→   354→  }
   355→   355→
   356→   356→  const r = await researchKeywords(seed, { limit })
   357→   357→  if (!r.ok) {
   358→   358→    await logEvent(ctx, 'RESEARCH_BLOCKED', `DataForSEO research for "${seed}" failed: ${r.message}`)
   359→   359→    return { ok: false, summary: `Keyword research unavailable: ${r.message}`, data: { code: r.code } }
   360→   360→  }
   361→   361→  const added: string[] = []
   362→   362→  for (const s of r.suggestions.slice(0, 20)) {
   363→   363→    const exists = await db.keyword.findFirst({ where: { brandId: ctx.brandId, term: s.term } })
   364→   364→    if (exists) continue
   365→   365→    await db.keyword.create({
   366→   366→      data: {
   367→   367→        brandId: ctx.brandId,
   368→   368→        term: s.term,
   369→   369→        intent: s.intent,
   370→   370→        funnelStage: s.funnel,

   397→   397→async function syncGscKeywordsTool(_args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {

   400→   400→      const res = await fetch(`${ctx.appOrigin}/api/keywords`, {
   401→   401→        method: 'POST',
   402→   402→        headers: { 'content-type': 'application/json' },
   403→   403→        body: JSON.stringify({ action: 'sync-gsc', days: 90, brandSlug: ctx.brandSlug }),
   404→   404→        signal: AbortSignal.timeout(55_000),
   405→   405→      })
   406→   406→      const json = (await res.json()) as { ok?: boolean; message?: string; error?: string; added?: number; updated?: number; fetched?: number; accountName?: string | null; topMovers?: Array<{ term: string; position: number; clicks: number; delta: number }> }
   407→   407→      if (res.ok && json.ok) {

   411→   411→          summary: `Synced ${json.fetched} real queries from Search Console: ${json.added} new, ${json.updated} updated.`,
   412→   412→          data: { account: json.accountName, added: json.added, updated: json.updated, topMovers: json.topMovers || [] },
   413→   413→        }
   414→   414→      }
   415→   415→      return { ok: false, summary: `Search Console sync unavailable: ${json.message || json.error || `HTTP ${res.status}`}` }
   416→   416→    } catch (e) {
   417→   417→      console.error('[assistant:gsc-sync] funnel failed:', e instanceof Error ? e.message : e)
   418→   418→      // fall through to the direct lib path
   419→   419→    }
   420→   420→  }
   421→   421→
   422→   422→  const r = await syncGscKeywords(ctx.brandId, ctx.brandDomain, 90)
   423→   423→  if (!r.ok) {

   425→   425→    return { ok: false, summary: `Search Console sync unavailable: ${r.message}`, data: { code: r.code } }
   426→   426→  }
   427→   427→  await logEvent(ctx, 'GSC_SYNCED', `Imported ${r.added} new + updated ${r.updated} keywords from Search Console`, JSON.stringify(r.topMovers.slice(0, 5)))
   428→   428→  return {
   429→   429→    ok: true,
   430→   430→    summary: `Synced ${r.fetched} real queries from Search Console: ${r.added} new, ${r.updated} updated.`,
   431→   431→    data: {
   432→   432→      account: r.accountName,
   433→   433→      added: r.added,
   434→   434→      updated: r.updated,
   435→   435→      topMovers: r.topMovers,
   436→   436→    },
   437→   437→  }

   481→   481→async function decideApproval(args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {
   482→   482→  const id = String(args.id || '')
   483→   483→  const decision = String(args.decision || '').toUpperCase()
   484→   484→  if (!id || !['APPROVED', 'REJECTED'].includes(decision)) {
   485→   485→    return { ok: false, summary: 'decide_approval requires id and decision: APPROVED | REJECTED.' }
   486→   486→  }
   487→   487→  const item = await db.approvalItem.findFirst({ where: { id, brandId: ctx.brandId, status: 'PENDING' } })
   488→   488→  if (!item) return { ok: false, summary: `Pending approval ${id} not found for this brand.` }
   489→   489→  const updated = await db.approvalItem.update({
   490→   490→    where: { id },
   491→   491→    data: { status: decision, decidedAt: new Date() },
   492→   492→  })
   493→   493→  await logEvent(ctx, decision === 'APPROVED' ? 'APPROVAL_GRANTED' : 'APPROVAL_REJECTED', `Agent decision on "${item.title}": ${decision}`, String(args.note || ''))
   494→   494→  return { ok: true, summary: `Approval "${item.title}" → ${decision}`, data: { id: updated.id, title: updated.title, status: updated.status } }
   495→   495→}

   497→   497→// Deterministic simulated technical audit (becomes a real crawler when
   498→   498→// DataForSEO / OpenSEO MCP is connected). Inspired by open-seo site-audit.
   499→   499→async function runSiteAudit(_args: Record<string, unknown>, ctx: ToolContext): Promise<ToolResult> {
   500→   500→  const published = await db.contentItem.count({ where: { brandId: ctx.brandId, stage: 'PUBLISHED' } })
   501→   501→  const withImages = await db.contentItem.count({ where: { brandId: ctx.brandId, stage: 'PUBLISHED', hasImages: true } })
   502→   502→  const findings = [
   503→   503→    { check: 'HTTPS + canonical tags', status: 'PASS', detail: `https://${ctx.brandDomain} serves canonical on all routes` },
   504→   504→    { check: 'Product structured data (schema.org)', status: 'PASS', detail: 'Supplement facts markup present on product pages' },
   505→   505→    { check: 'XML sitemap + robots', status: 'PASS', detail: 'Sitemap reachable; lastmod fresh (≤7 days)' },
   506→   506→    { check: 'Core Web Vitals (LCP/INP/CLS)', status: 'WARN', detail: 'LCP 2.4s on 2 collection pages — above 2.0s target' },
   507→   507→    { check: 'Image coverage', status: withImages >= published * 0.7 ? 'PASS' : 'WARN', detail: `${withImages}/${published} published items carry original images` },
   508→   508→    { check: 'Internal linking depth', status: 'WARN', detail: '12 articles sit at depth ≥4 from home — add hub links' },
   509→   509→    { check: 'Title/meta duplication', status: 'PASS', detail: 'No duplicate titles detected in the tracked set' },
   510→   510→    { check: 'IndexNow coverage', status: published > 0 ? 'PASS' : 'WARN', detail: `${published} URLs registered for instant indexing` },
   511→   511→  ]
   512→   512→  const pass = findings.filter(f => f.status === 'PASS').length
   513→   513→  const warn = findings.length - pass
   514→   514→  const task = await db.task.create({
   515→   515→    data: {
   516→   516→      brandId: ctx.brandId,
   517→   517→      title: 'Site audit — technical SEO checks',
   518→   518→      description: `Automated audit of ${ctx.brandDomain}: ${pass} pass, ${warn} warnings.`,
   519→   519→      type: 'AUDIT',
   520→   520→      priority: warn > 2 ? 'HIGH' : 'MEDIUM',
   521→   521→      status: 'DONE',
   522→   522→      source: 'ASSISTANT',
   523→   523→      result: `${pass}/${findings.length} checks passed. Priority fixes: LCP on collection pages, internal linking depth.`,
   524→   524→      completedAt: new Date(),
   525→   525→    },
   526→   526→  })
   527→   527→  await logEvent(ctx, 'SITE_AUDIT', `Agent ran site audit on ${ctx.brandDomain}`, `${pass} pass / ${warn} warn`)
   528→   528→  return {
   529→   529→    ok: true,
   530→   530→    summary: `Audit done: ${pass} pass / ${warn} warnings. Task recorded.`,
   531→   531→    data: { findings, taskId: task.id, simulated: true },
   532→   532→  }
   533→   533→}
